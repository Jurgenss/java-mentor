📝 String ▼

.length()

public int length()

Возвращает количество символов в строке.

String s = "Java";
System.out.println(s.length()); // 4

String empty = "";
System.out.println(empty.length()); // 0

String spaces = "   ";
System.out.println(spaces.length()); // 3 (пробелы тоже символы!)

.charAt(int index)

public char charAt(int index)

Возвращает символ по индексу (от 0 до length()-1). StringIndexOutOfBoundsException если индекс вне диапазона.

String s = "Java";
char c1 = s.charAt(0); // 'J'
char c2 = s.charAt(3); // 'a'
// char c3 = s.charAt(4); // ❌ StringIndexOutOfBoundsException

.substring(int beginIndex, int endIndex)

public String substring(int beginIndex, int endIndex)

Возвращает новую строку с beginIndex (включительно) по endIndex (не включительно). Исходная строка не меняется!

String s = "JavaRush";
String sub1 = s.substring(0, 4);  // "Java" (индексы 0,1,2,3)
String sub2 = s.substring(4);     // "Rush" (от 4 до конца)
System.out.println(s);            // "JavaRush" — не изменилась!

.toUpperCase() / .toLowerCase()

public String toUpperCase()
public String toLowerCase()

Возвращает новую строку в верхнем/нижнем регистре.

String s = "Java";
String upper = s.toUpperCase(); // "JAVA"
String lower = s.toLowerCase(); // "java"
System.out.println(s);          // "Java" — не изменилась!

.trim()

public String trim()

Убирает пробелы в начале и конце строки. Пробелы внутри не трогает. Возвращает новую строку.

String s = "  Java  ";
String trimmed = s.trim(); // "Java"
System.out.println(s);     // "  Java  " — не изменилась!

.startsWith(String prefix) / .endsWith(String suffix)

public boolean startsWith(String prefix)
public boolean endsWith(String suffix)

Проверяет, начинается/заканчивается ли строка на указанную подстроку.

String s = "JavaRush";
boolean b1 = s.startsWith("Java"); // true
boolean b2 = s.startsWith("Rush"); // false
boolean b3 = s.endsWith("Rush");   // true
boolean b4 = s.endsWith("Java");   // false

.contains(CharSequence s)

public boolean contains(CharSequence s)

Проверяет, содержится ли подстрока в строке.

String s = "JavaRush";
boolean b1 = s.contains("ava");  // true
boolean b2 = s.contains("Rush"); // true
boolean b3 = s.contains("Python"); // false

.replace(char oldChar, char newChar)

public String replace(char oldChar, char newChar)
public String replace(CharSequence target, CharSequence replacement)

Заменяет все вхождения. Для char — точно, для строки — как обычная замена.

String s = "Java";
String r1 = s.replace('a', 'o');      // "Jovo"
String r2 = s.replace("Ja", "Ko");    // "Kotlin"
// replaceAll — с регулярными выражениями:
String r3 = "a1b2c3".replaceAll("\\d", ""); // "abc"

.split(String regex)

public String[] split(String regex)

Разбивает строку на массив по разделителю. Разделитель — регулярное выражение!

String s = "яблоко,банан,груша";
String[] fruits = s.split(",");
// ["яблоко", "банан", "груша"]

String s2 = "a   b c"; // множественные пробелы
String[] parts = s2.split("\\s+"); // ["a", "b", "c"]

.toCharArray()

public char[] toCharArray()

Преобразует строку в массив символов.

String s = "Hi";
char[] chars = s.toCharArray();
// chars = ['H', 'i']
chars[0] = 'B';
String newStr = new String(chars); // "Bi"

.indexOf(String str) / .lastIndexOf(String str)

public int indexOf(String str)
public int lastIndexOf(String str)

Возвращает индекс первого/последнего вхождения. -1 если не найдено.

String s = "JavaJava";
int i1 = s.indexOf("Java");     // 0 (первое)
int i2 = s.lastIndexOf("Java"); // 4 (последнее)
int i3 = s.indexOf("Python");   // -1 (не найдено)

.isEmpty() / .isBlank()

public boolean isEmpty()
public boolean isBlank() // Java 11+

isEmpty(): true если длина = 0. isBlank(): true если пустая или только пробелы.

"".isEmpty();       // true
" ".isEmpty();      // false — пробел есть
" ".isBlank();      // true — только пробелы
"Java".isBlank();   // false

.equals(Object o) / .equalsIgnoreCase(String s)

public boolean equals(Object o)
public boolean equalsIgnoreCase(String s)

Сравнивает строки по содержимому, не по ссылкам. equalsIgnoreCase — без учёта регистра.

String a = "Java";
String b = "java";
a.equals(b);               // false
a.equalsIgnoreCase(b);     // true

// НЕЛЬЗЯ сравнивать через == !
String c = new String("Java");
a == c;                    // false — разные объекты!
a.equals(c);               // true — содержимое одинаковое

.concat(String str)

public String concat(String str)

Склеивает строки. Возвращает новую. Аналог +.

String s = "Hello".concat(" World"); // "Hello World"
// То же самое:
String s2 = "Hello" + " World";

🔧 StringBuilder ▼

.append(любой тип)

public StringBuilder append(...)

Добавляет в конец. Можно цепочкой. Возвращает тот же объект (не создаёт новый!).

StringBuilder sb = new StringBuilder();
sb.append("Hello");
sb.append(" ");
sb.append(2026);
// Цепочкой:
sb.append(" - ").append("Java").append(" Mentor");
String result = sb.toString(); // "Hello 2026 - Java Mentor"

.insert(int offset, ...)

public StringBuilder insert(int offset, ...)

Вставляет по указанному индексу. Остальное сдвигается вправо.

StringBuilder sb = new StringBuilder("Java");
sb.insert(1, "XX"); // вставить после 'J'
System.out.println(sb); // "JXXava"

.delete(int start, int end)

public StringBuilder delete(int start, int end)

Удаляет символы [start, end).

StringBuilder sb = new StringBuilder("JavaRush");
sb.delete(4, 8); // удалить "Rush" (индексы 4,5,6,7)
System.out.println(sb); // "Java"

.reverse()

public StringBuilder reverse()

Переворачивает содержимое.

StringBuilder sb = new StringBuilder("Java");
sb.reverse();
System.out.println(sb); // "avaJ"

.toString()

public String toString()

Преобразует обратно в String.

StringBuilder sb = new StringBuilder("Hello");
String s = sb.toString(); // "Hello"

📋 List (ArrayList / LinkedList) ▼

.add(E e) / .add(int i, E e)

boolean add(E e) | void add(int index, E e)

List<String> list = new ArrayList<>();
list.add("A");           // в конец → ["A"]
list.add("B");           // в конец → ["A", "B"]
list.add(1, "X");        // вставить по индексу → ["A", "X", "B"]

.get(int i)

E get(int index)

Получить элемент по индексу. ArrayList — O(1), LinkedList — O(n).

List<String> list = List.of("A", "B", "C");
String first = list.get(0); // "A"
String last = list.get(list.size() - 1); // "C"

.set(int i, E e)

E set(int index, E element)

Заменить элемент. Возвращает старое значение.

List<String> list = new ArrayList<>(List.of("A", "B", "C"));
String old = list.set(1, "X"); // old = "B", list = ["A", "X", "C"]

.remove(int i) / .remove(Object o)

E remove(int index) | boolean remove(Object o)

Внимание: для списка Integer remove(1) удалит по ИНДЕКСУ, не по значению! Для удаления по значению: remove(Integer.valueOf(1)).

List<String> list = new ArrayList<>(List.of("A", "B", "C"));
list.remove(1);       // удалили "B" → ["A", "C"]
list.remove("A");     // удалили по значению → ["C"]

🗺 Map (HashMap / TreeMap) ▼

.put(K key, V value)

V put(K key, V value)

Добавляет пару. Если ключ уже был — перезаписывает и возвращает старое значение. Если не было — возвращает null.

Map<String, Integer> map = new HashMap<>();
map.put("Java", 1995);       // null (не было)
Integer old = map.put("Java", 1996); // 1995 (перезаписали)
System.out.println(map); // {Java=1996}

.get(Object key) / .getOrDefault(Object key, V defaultValue)

V get(Object key) | V getOrDefault(Object key, V defaultValue)

get возвращает null если ключа нет. getOrDefault — указанное default-значение.

Map<String, String> map = Map.of("A", "Apple");
String val = map.get("A");          // "Apple"
String missing = map.get("B");      // null
String def = map.getOrDefault("B", "Unknown"); // "Unknown"

.keySet() / .values() / .entrySet()

Set<K> keySet() | Collection<V> values() | Set<Map.Entry<K,V>> entrySet()

Три способа обойти Map.

Map<String, Integer> map = new HashMap<>();
map.put("A", 1); map.put("B", 2);

// 1. По ключам
for (String key : map.keySet()) {
    System.out.println(key + " = " + map.get(key));
}

// 2. По entry (самый эффективный)
for (Map.Entry<String, Integer> e : map.entrySet()) {
    System.out.println(e.getKey() + " = " + e.getValue());
}

// 3. forEach
map.forEach((k, v) -> System.out.println(k + " = " + v));

🌊 Stream ▼

.filter(Predicate)

Stream<T> filter(Predicate<? super T> predicate)

Оставляет только элементы, для которых predicate возвращает true.

List<Integer> list = List.of(1, 2, 3, 4, 5, 6);
List<Integer> even = list.stream()
    .filter(x -> x % 2 == 0)     // 2, 4, 6
    .collect(Collectors.toList());
// [2, 4, 6]

.map(Function)

Stream<R> map(Function<? super T, ? extends R> mapper)

Преобразует каждый элемент в другой тип/значение.

List<String> names = List.of("java", "kotlin");
List<String> upper = names.stream()
    .map(String::toUpperCase)    // "JAVA", "KOTLIN"
    .collect(Collectors.toList());
// ["JAVA", "KOTLIN"]

.flatMap(Function)

Stream<R> flatMap(Function<T, Stream<R>> mapper)

"Схлопывает" вложенные структуры в один плоский поток.

List<List<String>> nested = List.of(
    List.of("A", "B"),
    List.of("C", "D")
);
List<String> flat = nested.stream()
    .flatMap(List::stream)       // "A", "B", "C", "D"
    .collect(Collectors.toList());
// ["A", "B", "C", "D"]

.collect(Collectors.toList()) / .collect(Collectors.joining())

R collect(Collector collector)

Собирает элементы в коллекцию, строку, Map — зависит от коллектора.

// В список
List<String> list = stream.collect(Collectors.toList());

// В строку через запятую
String joined = stream.collect(Collectors.joining(", "));

// Группировка
Map<String, List<User>> byCity = users.stream()
    .collect(Collectors.groupingBy(User::getCity));

.reduce(accumulator)

Optional<T> reduce(BinaryOperator<T> accumulator)

Свёртка: сумма, произведение, максимум и т.д.

List<Integer> numbers = List.of(1, 2, 3, 4, 5);

// Сумма
int sum = numbers.stream().reduce(0, (a, b) -> a + b); // 15

// Максимум
Optional<Integer> max = numbers.stream().reduce(Integer::max);

// Альтернативы:
int sum2 = numbers.stream().mapToInt(Integer::intValue).sum();
Optional<Integer> max2 = numbers.stream().max(Integer::compareTo);

🔢 Math (статические методы) ▼

Math.abs() / Math.max() / Math.min()

int abs = Math.abs(-5);      // 5
double abs2 = Math.abs(-3.14); // 3.14

int max = Math.max(10, 20);    // 20
double max2 = Math.max(3.5, 2.1); // 3.5

int min = Math.min(10, 20);    // 10

Math.pow() / Math.sqrt()

double pow = Math.pow(2, 3);   // 8.0 (2 в степени 3)
double sqrt = Math.sqrt(9);     // 3.0
double sqrt2 = Math.sqrt(2);    // 1.4142...

Math.round() / Math.ceil() / Math.floor()

long r1 = Math.round(3.2);  // 3
long r2 = Math.round(3.7);  // 4

double c1 = Math.ceil(3.1);  // 4.0  (вверх)
double c2 = Math.ceil(3.9);  // 4.0

double f1 = Math.floor(3.1); // 3.0  (вниз)
double f2 = Math.floor(3.9); // 3.0

Math.random()

double r = Math.random(); // [0.0, 1.0)
// Случайное целое 0..99:
int randomInt = (int) (Math.random() * 100);
// Случайное целое 1..100:
int random1to100 = (int) (Math.random() * 100) + 1;

🎁 Optional ▼

.of() / .ofNullable() / .empty()

Optional<String> opt1 = Optional.of("Java");       // значение есть, не null
// Optional<String> opt2 = Optional.of(null);        // ❌ NullPointerException!

Optional<String> opt2 = Optional.ofNullable(null);   // ✅ можно null → Optional.empty
Optional<String> opt3 = Optional.empty();            // явно пустой

.isPresent() / .get()

Optional<String> opt = Optional.of("Java");
if (opt.isPresent()) {
    String s = opt.get(); // "Java"
}
// get() на пустом Optional → NoSuchElementException!

.orElse() / .orElseGet() / .orElseThrow()

Optional<String> opt = Optional.empty();

String s1 = opt.orElse("default");              // "default"
String s2 = opt.orElseGet(() -> "computed");    // "computed" (Supplier вызывается только при empty)
// String s3 = opt.orElseThrow();               // NoSuchElementException
String s4 = opt.orElseThrow(() -> new RuntimeException("Нет значения!"));

.map() / .filter()

Optional<String> opt = Optional.of("Java");

Optional<Integer> len = opt.map(String::length); // Optional[4]
Optional<String> filtered = opt.filter(s -> s.startsWith("J")); // Optional["Java"]
Optional<String> empty = opt.filter(s -> s.startsWith("X"));    // Optional.empty

🛠 Collections (утилитный класс) ▼

Collections.sort() / Collections.reverse()

List<String> list = new ArrayList<>(List.of("C", "A", "B"));

Collections.sort(list);                    // [A, B, C] — natural order
Collections.sort(list, Comparator.reverseOrder()); // [C, B, A]
Collections.reverse(list);                 // перевернуть порядок

Collections.min() / Collections.max()

List<Integer> list = List.of(5, 2, 9, 1);
int min = Collections.min(list); // 1
int max = Collections.max(list); // 9

Collections.frequency()

List<String> list = List.of("A", "B", "A", "C", "A");
int freq = Collections.frequency(list, "A"); // 3

Collections.unmodifiableList() / Collections.synchronizedList()

List<String> original = new ArrayList<>(List.of("A", "B"));
List<String> unmod = Collections.unmodifiableList(original);
// unmod.add("C"); // ❌ UnsupportedOperationException

List<String> sync = Collections.synchronizedList(new ArrayList<>());
