🌐 Spring MVC

Spring MVC — основа веб-приложений на Spring. Даже для джуна важно понимать, как обрабатывается HTTP-запрос и какие аннотации за что отвечают.

Как Spring MVC обрабатывает HTTP-запрос

1. Запрос приходит на DispatcherServlet (фронт-контроллер)
2. DispatcherServlet спрашивает HandlerMapping: "Кто обрабатывает GET /users?"
3. HandlerMapping находит нужный метод в @Controller
4. DispatcherServlet вызывает этот метод
5. Метод возвращает ModelAndView (или объект + @ResponseBody)
6. ViewResolver находит HTML-шаблон (или объект сериализуется в JSON)
7. Ответ уходит клиенту
DispatcherServlet — центральный сервлет Spring MVC. Принимает все HTTP-запросы и распределяет их по контроллерам через HandlerMapping. Контроллер возвращает ModelAndView, ViewResolver находит представление. Для REST API (@RestController) объект сразу сериализуется в JSON.

Полный CRUD-контроллер (REST API)

@RestController
@RequestMapping("/api/users")
public class UserController {

    @Autowired
    private UserService userService;

    // GET /api/users — все пользователи
    @GetMapping
    public List<User> getAll() {
        return userService.findAll();
    }

    // GET /api/users/123 — один пользователь
    @GetMapping("/{id}")
    public User getById(@PathVariable Long id) {
        return userService.findById(id);
    }

    // POST /api/users — создать нового
    @PostMapping
    @ResponseStatus(HttpStatus.CREATED) // 201 Created
    public User create(@RequestBody @Valid User user) {
        return userService.create(user);
    }

    // PUT /api/users/123 — обновить целиком
    @PutMapping("/{id}")
    public User update(@PathVariable Long id, @RequestBody @Valid User user) {
        return userService.update(id, user);
    }

    // PATCH /api/users/123 — частичное обновление
    @PatchMapping("/{id}")
    public User patch(@PathVariable Long id, @RequestBody Map<String, Object> fields) {
        return userService.patch(id, fields);
    }

    // DELETE /api/users/123 — удалить
    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT) // 204 No Content
    public void delete(@PathVariable Long id) {
        userService.delete(id);
    }

    // GET /api/users/search?name=Ivan — поиск по параметру
    @GetMapping("/search")
    public List<User> search(@RequestParam String name) {
        return userService.findByName(name);
    }

    // GET /api/users?page=0&size=10 — пагинация
    @GetMapping(params = {"page", "size"})
    public Page<User> getPage(@RequestParam int page, @RequestParam int size) {
        return userService.findPage(page, size);
    }
}

@RequestParam vs @PathVariable vs @RequestBody

Откуда берёт данныеПример запроса
@PathVariable Из URL-пути GET /users/123 → @PathVariable Long id
@RequestParam Из query-параметров URL GET /users?name=Ivan → @RequestParam String name
@RequestBody Из тела запроса (JSON) POST /users + JSON в теле → @RequestBody User user 
// URL: POST /api/users/123/orders?status=NEW
// Тело: {"amount": 500.00}

Валидация входящих данных (@Valid)

// DTO с аннотациями валидации
public class CreateUserRequest {

    @NotBlank(message = "Имя обязательно")
    @Size(min = 2, max = 50, message = "Имя должно быть от 2 до 50 символов")
    private String name;

    @Email(message = "Некорректный email")
    @NotBlank(message = "Email обязателен")
    private String email;

    @Min(value = 18, message = "Возраст должен быть >= 18")
    @Max(value = 150, message = "Возраст должен быть <= 150")
    private int age;
}

// В контроллере
@PostMapping
@ResponseStatus(HttpStatus.CREATED)
public User create(@RequestBody @Valid CreateUserRequest request) {
    return userService.create(request);
}

// Обработка ошибок валидации
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(MethodArgumentNotValidException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST) // 400
    public Map<String, String> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, String> errors = new HashMap<>();
        ex.getBindingResult().getFieldErrors()
            .forEach(e -> errors.put(e.getField(), e.getDefaultMessage()));
        return errors;
    }
}

ResponseEntity — полный контроль над ответом

@GetMapping("/{id}")
public ResponseEntity<User> getById(@PathVariable Long id) {
    try {
        User user = userService.findById(id);
        return ResponseEntity.ok(user); // 200
    } catch (UserNotFoundException e) {
        return ResponseEntity.notFound().build(); // 404
    }
}

@PostMapping
public ResponseEntity<User> create(@RequestBody @Valid User user) {
    User created = userService.create(user);
    URI location = ServletUriComponentsBuilder
        .fromCurrentRequest()
        .path("/{id}")
        .buildAndExpand(created.getId())
        .toUri();
    return ResponseEntity.created(location).body(created); // 201 + заголовок Location
}

Model и ModelAndView (для Thymeleaf/JSP)

@Controller // не @RestController!
@RequestMapping("/web/users")
public class UserWebController {

    @GetMapping
    public String listUsers(Model model) {
        List<User> users = userService.findAll();
        model.addAttribute("users", users);      // передаём данные
        model.addAttribute("title", "Пользователи");
        return "users/list";                     // thymeleaf-шаблон: users/list.html
    }

    @GetMapping("/{id}")
    public ModelAndView viewUser(@PathVariable Long id) {
        ModelAndView mav = new ModelAndView("users/view");
        mav.addObject("user", userService.findById(id));
        return mav;
    }
}

FilterInterceptor
УровеньСервлет-контейнер (до DispatcherServlet)Spring (после DispatcherServlet)
Доступ кServletRequest, ServletResponseHttpServletRequest, HttpServletResponse, Handler
КогдаЛогирование всех запросов, CORS, сжатиеАутентификация, логирование методов
// Interceptor
@Component
public class LoggingInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest request, 
                             HttpServletResponse response, 
                             Object handler) {
        System.out.println("→ " + request.getMethod() + " " + request.getRequestURI());
        return true; // false = прервать запрос
    }

    @Override
    public void afterCompletion(HttpServletRequest request,
                                 HttpServletResponse response,
                                 Object handler, Exception ex) {
        System.out.println("← " + response.getStatus());
    }
}

// Регистрация
@Configuration
public class WebConfig implements WebMvcConfigurer {
    @Autowired
    private LoggingInterceptor loggingInterceptor;

    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        registry.addInterceptor(loggingInterceptor)
            .addPathPatterns("/api/**")         // ко всем API
            .excludePathPatterns("/api/health"); // кроме health-check
    }
}

Настройка CORS

// Глобально
@Configuration
public class CorsConfig implements WebMvcConfigurer {
    @Override
    public void addCorsMappings(CorsRegistry registry) {
        registry.addMapping("/api/**")
            .allowedOrigins("http://localhost:3000") // фронтенд
            .allowedMethods("GET", "POST", "PUT", "DELETE")
            .allowedHeaders("*")
            .allowCredentials(true);
    }
}
